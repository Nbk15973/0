import numpy as np
import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import LSTM, Dense, Input
import random
path = tf.keras.utils.get_file("nietzsche.txt","https://s3.amazonaws.com/text-datasets/nietzsche.txt")
text = open(path, "r", encoding="utf-8").read().lower()
text = text[:50000]
chars = sorted(set(text))
char_to_idx = {c:i for i, c in enumerate(chars)}
idx_to_char = {i:c for i, c in enumerate(chars)}
maxlen = 40
step = 3
X, y = [], []
for i in range(0, len(text)-maxlen, step):
 X.append([char_to_idx[c] for c in text[i:i+maxlen]])
 y.append(char_to_idx[text[i+maxlen]])
X = np.array(X)
y = tf.keras.utils.to_categorical(y, num_classes=len(chars))
model = Sequential([
Input(shape=(maxlen, 1)),
LSTM(128),
Dense(len(chars), activation="softmax")
])
model.compile(loss="categorical_crossentropy", optimizer="adam")
X = X.reshape((X.shape[0], X.shape[1], 1)) / float(len(chars))
model.fit(X, y, epochs=2, batch_size=128)
start = random.randint(0, len(X)-1)
sentence = X[start]
generated = ""
for i in range(200):
 pred = model.predict(sentence.reshape(1, maxlen, 1), verbose=0)[0]
 next_index = np.random.choice(len(pred), p=pred)
 next_char = idx_to_char[next_index]
 generated += next_char
 sentence = np.append(sentence[1:], [[next_index / float(len(chars))]], axis=0)
print("Generated Text:\n")
print(generated)

import tensorflow as tf
from tensorflow.keras.preprocessing.sequence import pad_sequences
from tensorflow.keras.utils import to_categorical
import numpy as np
sentences = [["I","love","NLP"],["She","reads","books"],["He","plays","football"]]
tags = [["PRON","VERB","NOUN"],["PRON","VERB","NOUN"],["PRON","VERB","NOUN"]]
word2idx = {"I":1,"love":2,"NLP":3,"She":4,"reads":5,"books":6,"He":7,"plays":8,"football":9}
tag2idx = {"PRON":0,"VERB":1,"NOUN":2}
idx2tag = {0:"PRON",1:"VERB",2:"NOUN"}
X = [[word2idx[w] for w in s] for s in sentences]
y = [[tag2idx[t] for t in tag] for tag in tags]
X = pad_sequences(X, maxlen=5, padding='post')
y = pad_sequences(y, maxlen=5, padding='post')
y = [to_categorical(i, num_classes=3) for i in y]
y = np.array(y)
model = tf.keras.Sequential([
tf.keras.layers.Embedding(10, 16, input_length=5),
tf.keras.layers.LSTM(32, return_sequences=True),
tf.keras.layers.TimeDistributed(tf.keras.layers.Dense(3, activation='softmax'))
])
model.compile(optimizer='adam',loss='categorical_crossentropy',metrics=['accuracy'])
model.fit(X, y, epochs=10, batch_size=2)
test_sentence = ["I", "play", "football"]
test = [[word2idx.get(w,0) for w in test_sentence]]
test = pad_sequences(test, maxlen=5, padding='post')
pred = model.predict(test)
pred_tags = [idx2tag[np.argmax(p)] for p in pred[0]]
print("Sentence:", test_sentence)
print("Predicted POS Tags:", pred_tags[:len(test_sentence)])

Simple MoCo Program

Simple MoCo Program

import torch, torch.nn as nn
x=torch.randn(32,3,32,32)
class E(nn.Module):
 def __init__(s):
  super().__init__()
  s.m=nn.Sequential(
  nn.Flatten(),
  nn.Linear(3*32*32,128))
 def forward(s,x):
  return s.m(x)
q=E()
k=E()
loss_fn=nn.CosineEmbeddingLoss()
opt=torch.optim.Adam(q.parameters(),lr=0.001)
for e in range(3):
 a=q(x)
 b=k(x)
 y=torch.ones(a.size(0))
 loss=loss_fn(a,b,y)
 opt.zero_grad()
 loss.backward()
 opt.step()
 print("Epoch:",e,"Loss:",round(loss.item(),2))
print("Feature extractor trained using MoCo representation learning.")
